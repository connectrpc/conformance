# connect-go accepts a timeout with a leading "+".
Connect Timeout Format Requests/**/leading-plus
Connect GET Timeout Format Requests/**/leading-plus
# connect-go does not enforce deadlines on the server, so a zero timeout
# succeeds instead of failing with deadline_exceeded.
Connect Timeout Format Requests/**/zero
Connect GET Timeout Format Requests/**/zero
# connect-go treats an empty timeout header as if it were absent.
Connect Timeout Format Requests/**/empty
Connect GET Timeout Format Requests/**/empty
